## 5. Schema.org Structured Data (Rich Snippets)

The platform implements 5 primary Schema.org JSON-LD microdata formats:

### 1. Global WebSite & SearchAction Schema
```json
{
  "@context": "https://schema.org",
  "@graph": [
    {
      "@type": "Organization",
      "@id": "https://web.digitalclassworld.in/#organization",
      "name": "Digital Class",
      "url": "https://web.digitalclassworld.in",
      "logo": "https://web.digitalclassworld.in/images/logo.png",
      "sameAs": [
        "https://www.facebook.com/digitalclassworld",
        "https://www.instagram.com/digitalclassworld",
        "https://www.linkedin.com/company/digitalclassworld"
      ]
    },
    {
      "@type": "WebSite",
      "@id": "https://web.digitalclassworld.in/#website",
      "url": "https://web.digitalclassworld.in",
      "name": "Digital Class",
      "publisher": { "@id": "https://web.digitalclassworld.in/#organization" },
      "potentialAction": {
        "@type": "SearchAction",
        "target": "https://web.digitalclassworld.in/search?q={search_term_string}",
        "query-input": "required name=search_term_string"
      }
    }
  ]
}
```

```json
  "logo": "{logo_url}",
  "telephone": "{contact_number}",
  "email": "{business_email}",
  "address": {
    "@type": "PostalAddress",
    "streetAddress": "{address}",
    "addressLocality": "{city}",
    "addressRegion": "{state}",
    "postalCode": "{pincode}",
    "addressCountry": "IN"
  }
}
```

### 3. Course Schema (For Course Details Pages)
```json
{
  "@context": "https://schema.org",
  "@type": "Course",
  "name": "{course_title}",
  "description": "{short_description}",
  "provider": {
    "@type": "Organization",

    "sameAs": "https://web.digitalclassworld.in"
  },
  "offers": {
    "@type": "Offer",
    "price": "{price}",
    "priceCurrency": "INR",
    "availability": "https://schema.org/InStock",
    "category": "Educational Course"
  },
  "hasCourseInstance": {
    "@type": "CourseInstance",
    "courseMode": "Online",
    "courseWorkload": "PT{duration}H"
  }
}
```
